\begin{quote}
\textit{Summary of the theory-prediction grid stored in
\code{theory-data/output8/}, \code{theory-data/output13/} and consolidated
into \code{theory-data/analysis/converted\_data.json}. Restricted to the
$\rho$-related differential observables, inclusive cross sections,
parameters and PDF variations; non-$\rho$ differential distributions are
omitted.}
\end{quote}

\section*{Grid dimensions}

\begin{itemize}
  \item \textbf{Energies:} 8~TeV and 13~TeV.
  \item \textbf{Orders in the converted JSON:} \code{LO}, \code{NLO},
        \code{NNLO\_noVVF} (the full physical NNLO prediction). Raw bare
        \code{NNLO} $\epsilon$-pole XMLs also exist on disk at both energies
        but are used only as an internal pole-cancellation check, not shipped
        in the converted dataset.
  \item \textbf{Top-quark mass grid:} $167.5,\ 171.5,\ 172.5,\ 173.5,\
        177.5$~GeV, for every order and energy.
  \item \textbf{Scale variations:} 3 central scale choices ($H_T$, $H_T/2$,
        $H_T/4$), each with the standard 7-point $(\mu_R,\mu_F)$ envelope,
        consolidated onto 13 distinct $(\mu_R,\mu_F)$ points per mass/order.
        Nominal central scale is $H_T/2$.
  \item \textbf{Inclusive cross section:} $\epsilon$-expansion coefficients
        $[c_0, c_{-1}, c_{-2}, c_{-3}, c_{-4}]$ from the STRIPPER
        dimensional-regularization output; $c_0$ (pb) is the physical cross
        section, $c_{-1}\ldots c_{-4}$ are IR-pole coefficients kept as a
        numerical cancellation check.
\end{itemize}

\noindent \code{rhoatlas8} and \code{rhoatlas13} are stored at \emph{both}

collision energy); \code{rhocms} is energy-tagged instead. $\rho$ is the
top-mass-sensitive jet observable of Alioli, Fernandez, Fuster, Irles, Moch,
Uwer, Vos, \emph{A new observable to measure the top-quark mass at hadron
colliders}, Eur.\ Phys.\ J.\ C 73 (2013) 2438, arXiv:1303.6415.
